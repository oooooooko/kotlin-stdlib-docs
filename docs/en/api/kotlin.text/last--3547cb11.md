fun main() {
    val string = "Kotlin 1.4.0"
    println(string.last())
    println(string.last { it.isLetter() })
    println(string.lastOrNull { it > 'z' })
    assertFails { string.last { it > 'z' } }

    val emptyString = ""
    println(emptyString.lastOrNull())
    assertFails { emptyString.last() }
}
