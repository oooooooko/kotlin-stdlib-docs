fun main() {
    val list = listOf(1, 2, 3, 4)
    println(list.last())
    println(list.last { it % 2 == 1 })
    println(list.lastOrNull { it < 0 })
    assertFails { list.last { it < 0 } }

    val emptyList = emptyList<Int>()
    println(emptyList.lastOrNull())
    assertFails { emptyList.last() }
}
